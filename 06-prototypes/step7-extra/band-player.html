🎸 F7 — The band that follows you

Synthesized original backing (drums + root–fifth bass + chord stabs). The band sits in your last practice tempo, not a fixed click. No audio leaves the device (Ban 5).

Press play to hear the band at your tempo.
const audioCtx = new (window.AudioContext || window.webkitAudioContext)(); let node = null, srcBuf = null; function buildBuffer(bpm, beats, feel, cycle, lessonDefault) { const chords = { E7: { frets: [0,2,0,1,0,0] }, A7: { frets: [null,0,2,0,2,0] }, B7: { frets: [null,2,1,2,0,2] }, C: { frets: [null,3,2,0,1,0] }, G: { frets: [3,2,0,0,0,3] }, D: { frets: [null,3,2,0,0,0] }, Am: { frets: [null,0,2,2,1,0] }, Em: { frets: [0,2,2,0,0,0] } }; const cycleArr = cycle.split(',').map(s => s.trim()).filter(Boolean); const r = window.BandEngine.buildBand({ chordCycle: cycleArr, chords, bpm: bpm || null, lessonDefaultBpm: lessonDefault, beats: beats, feel: feel, bars: cycleArr.length, seed: 7 }); return r; } function floatToAudio(f32, sr) { const ab = audioCtx.createBuffer(1, f32.length, sr); ab.copyToChannel(f32, 0); return ab; } function play() { stop(); const bpm = parseFloat(document.getElementById('bpm').value); const beats = parseInt(document.getElementById('beats').value, 10); const feel = document.getElementById('feel').value; const cycle = document.getElementById('cycle').value; const lessonDefault = parseFloat(document.getElementById('lessonDefault').value); const r = buildBuffer(isNaN(bpm) ? null : bpm, beats, feel, cycle, lessonDefault); srcBuf = floatToAudio(r.buffer, r.sampleRate); node = audioCtx.createBufferSource(); node.buffer = srcBuf; node.connect(audioCtx.destination); node.start(); document.getElementById('out').textContent = 'Playing at BPM ' + r.bpm + ' (' + (isNaN(bpm) ? 'lesson default ' + lessonDefault : 'your practice tempo') + ')\n' + 'loop length ' + r.durationSec.toFixed(2) + 's · feel ' + r.feel + ' · bars ' + r.bars + '\n' + 'stems: ' + r.nominal.length + ' audible events\n' + 'bass root ' + (r.nominal.find(n => n.stem === 'bass') || {}).note; } function stop() { if (node) { try { node.stop(); } catch (e) {} node = null; } } document.getElementById('play').onclick = function () { audioCtx.resume(); play(); }; document.getElementById('stop').onclick = stop; document.getElementById('sample').onclick = function () { document.getElementById('bpm').value = 58; document.getElementById('beats').value = 4; document.getElementById('feel').value = 'swing'; document.getElementById('cycle').value = 'E7,A7,B7'; document.getElementById('lessonDefault').value = 68; audioCtx.resume(); play(); };
